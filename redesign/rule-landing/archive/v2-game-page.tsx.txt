  const rootRef = useRef<HTMLDivElement>(null);
  useReveal(rootRef);
  const [morphed] = useState(isOpeningRule);
  // Arriving by morph, start at the top now so the title flies into a hero that is on screen.
  useLayoutEffect(() => {
    if (morphed) scrollTo({ top: 0, behavior: "instant" });
  }, [morphed]);
  // The card's title fades out early in the morph, so the live, scrambling heading carries it.
  useLayoutEffect(() => {
    const root = document.documentElement;
    root.dataset.proxTitle = "";
    return () => {
      delete root.dataset.proxTitle;
    };
  }, []);

  const [round, setRound] = useState(0);
  const [results, setResults] = useState<Results>(OPEN);
  const [solveRun, setSolveRun] = useState(0);
  const done = results.every((stars) => stars !== null);

    if (done) {
      setResults(OPEN);
      setSolveRun(0);
      setRound((value) => value + 1);
      return;

    setSolveRun((value) => value + 1);
    const open = results.filter((stars) => stars === null).length;
    setTimeout(
      () => {
        const end = document.getElementById("results");

      open * SOLVE_STEP + 520,

  }

  useEffect(() => {

  const openOrder = results.map((stars, index) =>
    stars === null ? results.slice(0, index).filter((s) => s === null).length : -1,
  );
  const props = { results, done, onFinish: finishAll };
  // The heading resolves after the card's title has landed when it flew in, sooner otherwise.
  const titleDelay = morphed ? 160 : 120;

  return (
    <div

// ---------- Scramble: each character resolves out of noise, left to right ----------

// Noise is drawn from glyphs about as wide as the real letter, so a thin "i" never wears a "W".
const NOISE = [
  [/[iljI1!|.,:;']/, "il1!|:;"],
  [/[tfr]/, "tfrjc"],
  [/[mwMW@%&]/, "MW%&@"],
  [/[A-Z]/, "ABCDEGHKNOPRSUVXYZ"],
] as const;
const LOWER = "abdeghknopqsuvxyz02345689";
const noiseFor = (char: string) => NOISE.find(([pattern]) => pattern.test(char))?.[1] ?? LOWER;
const glyph = (char: string) => {
  const set = noiseFor(char);
  return set[Math.floor(Math.random() * set.length)] ?? "#";
};
// The first frame's noise is fixed, so the server's markup and the first client render agree.
const seededGlyph = (char: string, index: number) => {
  const set = noiseFor(char);
  return set[(index * 7 + 3) % set.length] ?? "#";
};

type ScrambleProps = {
  text: string;
  /** False holds the text as still noise. */
  shown: boolean;
  /** Decode on mount instead of starting resolved. */
  intro?: boolean;
  delay?: number;
  /** ms between one character resolving and the next. */
  step?: number;
};

/** Every glyph keeps the real character's box (drawn transparent) and the noise sits over it,
 * so the line never reflows while it decodes. */
function Scramble({ text, shown, intro = false, delay = 0, step = 34 }: ScrambleProps) {
  const ref = useRef<HTMLSpanElement>(null);
  const [noisy] = useState(() => !shown || intro);

  useLayoutEffect(() => {
    const chars = [...(ref.current?.querySelectorAll<HTMLElement>(".ch") ?? [])];
    if (!shown) {
      for (const ch of chars) ch.dataset.g = glyph(ch.textContent ?? "");
      return;
    }
    if (reducedMotion()) {
      for (const ch of chars) delete ch.dataset.g;
      return;
    }
    if (!chars.some((ch) => ch.dataset.g)) return;
    let frame = 0;
    let rolled = 0;
    const start = performance.now() + delay;
    const tick = (now: number) => {
      const roll = now - rolled > 45;
      if (roll) rolled = now;
      let left = 0;
      chars.forEach((ch, index) => {
        if (!ch.dataset.g) return;
        if (now - start >= index * step + 120) {
          delete ch.dataset.g;
          return;
        }
        left += 1;
        if (roll) ch.dataset.g = glyph(ch.textContent ?? "");
      });
      if (left) frame = requestAnimationFrame(tick);
    };
    frame = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(frame);
  }, [shown, delay, step]);

  let index = 0;
  return (
    <span
      ref={ref}
      className="scr"
      data-intro={intro || undefined}
      aria-hidden={!shown || undefined}
    >
      {[...text].map((char, at) =>
        char === " " ? (
          " "
        ) : (
          <span key={at} className="ch" data-g={noisy ? seededGlyph(char, index++) : undefined}>
            {char}
          </span>
        ),
      )}
    </span>
  );
}

// ---------- Progress: the hero's level select, the HUD and the rail ----------

function Stars({ stars }: { stars: number | null }) {
  if (stars === null) return null;
  if (stars === 0) return <span className="stars skip">skipped</span>;
  return (
    <span className="stars" aria-label={`${stars} of 3 stars`}>
      {[1, 2, 3].map((at) => (
        <i key={at} data-on={at <= stars || undefined}>
          ★
        </i>
      ))}
    </span>
  );
}

const starTotal = (results: Results) =>
  results.reduce<number>((sum, stars) => sum + (stars ?? 0), 0);
const clearedCount = (results: Results) => results.filter((stars) => stars !== null).length;

function LevelSelect({ results }: { results: Results }) {
  return (
    <nav className="lv-select panel in" style={{ "--i": 4 }} aria-label="Level select">
      <div className="lv-select-head">
        <span>Level select</span>
        <span>
          {clearedCount(results)}/{LEVELS.length} cleared
        </span>
      </div>
      <ol>
        {LEVELS.map((level, index) => {
          const stars = results[index] ?? null;
          return (
            <li key={level.name}>
              <a href={`#lv-${index + 1}`} data-clear={stars !== null || undefined}>
                <b>LV{index + 1}</b>
                <span className="nm">
                  {stars === null && (
                    <span className="sr-only">Level {index + 1}, not cleared</span>
                  )}
                  <Scramble text={level.name} shown={stars !== null} step={24} />
                </span>
                <Stars stars={stars} />
              </a>
            </li>
          );
        })}
      </ol>
    </nav>
  );
}

/** The levels hang off a rail whose fill reaches the last level of the cleared run from LV1. */
function Rail({ results, children }: { results: Results; children: ReactNode }) {
  const railRef = useRef<HTMLDivElement>(null);
  const fillRef = useRef<HTMLElement>(null);
  const run = results.findIndex((stars) => stars === null);
  const reached = run === -1 ? results.length : run;

  useLayoutEffect(() => {
    const rail = railRef.current;
    const fill = fillRef.current;
    if (!rail || !fill) return;
    const place = () => {
      const nodes = rail.querySelectorAll<HTMLElement>(".node");
      const last = nodes[reached - 1];
      const track = fill.parentElement ?? rail;
      const top = track.getBoundingClientRect().top;
      const height = track.offsetHeight || 1;
      // Down to the centre of the last cleared node.
      const y =
        reached === results.length
          ? height
          : last
            ? last.getBoundingClientRect().top + last.offsetHeight / 2 - top
            : 0;
      fill.style.transform = `scaleY(${Math.min(1, y / height)})`;
    };
    place();
    const observer = new ResizeObserver(place);
    observer.observe(rail);
    return () => observer.disconnect();
  }, [reached, results.length]);

  return (
    <div ref={railRef} className="lvls">
      <span className="rail" aria-hidden="true">
        <i ref={fillRef} />
      </span>
      {children}
    </div>
  );
}

// ---------- One level: carry the piece to where it lives ----------

type LevelProps = {
  n: number;
  /** Remounted by Replay: already seen, so it skips the reveal. */
  replayed: boolean;
  level: Level;
  stars: number | null;
  /** Set by "Finish all levels": solve this level after that many ms. */
  solveAt: number | null;
  onClear: (stars: number) => void;
};

type Drag = { x0: number; y0: number; lastX: number; lastT: number; rot: number; moved: boolean };

function LevelBoard({ n, replayed, level, stars, solveAt, onClear }: LevelProps) {
  const slots = level.rows.flatMap((row) => ("slot" in row ? [row.slot] : []));
  const [placed, setPlaced] = useState<string | null>(null);
  const [miss, setMiss] = useState<{ id: string; at: number } | null>(null);
  const [mistakes, setMistakes] = useState(0);
  const [picked, setPicked] = useState(false);
  const [dragging, setDragging] = useState(false);
  const [near, setNear] = useState<string | null>(null);

  const pieceRef = useRef<HTMLButtonElement>(null);
  const seatRef = useRef<HTMLSpanElement>(null);
  const slotRefs = useRef(new Map<string, HTMLElement>());
  const dragRef = useRef<Drag | null>(null);
  const nearRef = useRef<string | null>(null);
  const flipFrom = useRef<{ rect: DOMRect; rot: number } | null>(null);

  const cleared = placed !== null;
  const said = placed
    ? slots.find((slot) => slot.id === placed)
    : miss
      ? slots.find((slot) => slot.id === miss.id)
      : undefined;

  function flyHome(transform: string) {
    const piece = pieceRef.current;
    if (!piece) return;
    piece.style.transform = "";
    if (!transform || reducedMotion()) return;
    piece.animate([{ transform }, { transform: "none" }], { duration: 360, easing: OUT });
  }

  function drop(id: string, solved = false) {
    const piece = pieceRef.current;
    if (!piece || cleared) return;
    setPicked(false);
    if (id === level.answer) {
      flipFrom.current = { rect: piece.getBoundingClientRect(), rot: dragRef.current?.rot ?? 0 };
      piece.style.transform = "";
      setPlaced(id);
      setMiss(null);
      onClear(solved ? 0 : Math.max(1, 3 - mistakes));
      return;
    }
    setMistakes((value) => value + 1);
    setMiss({ id, at: performance.now() });
    flyHome(piece.style.transform);
  }

  // Finish all levels: drop the piece on the answer, as if carried there.
  const dropRef = useRef(drop);
  dropRef.current = drop;
  useEffect(() => {
    if (solveAt === null || stars !== null) return;
    const timer = setTimeout(() => dropRef.current(level.answer, true), solveAt);
    return () => clearTimeout(timer);
  }, [solveAt, stars, level.answer]);

  // The piece flies from wherever it was let go into its seat in the tree.
  useLayoutEffect(() => {
    const from = flipFrom.current;
    const seat = seatRef.current;
    flipFrom.current = null;
    if (!from || !seat || reducedMotion()) return;
    const to = seat.getBoundingClientRect();
    const dx = from.rect.left - to.left;
    const dy = from.rect.top - to.top;
    seat.animate(
      [
        { transform: `translate(${dx}px, ${dy}px) rotate(${from.rot}deg) scale(1.05)` },
        { transform: "none" },
      ],
      { duration: 420, easing: SNAP },
    );
  }, [placed]);

  function nearest(x: number, y: number) {
    let best: string | null = null;
    let bestDistance = 48;
    for (const [id, element] of slotRefs.current) {
      const box = element.getBoundingClientRect();
      const dx = Math.max(box.left - x, 0, x - box.right);
      const dy = Math.max(box.top - y, 0, y - box.bottom);
      const distance = Math.hypot(dx, dy);
      if (distance < bestDistance) {
        bestDistance = distance;
        best = id;
      }
    }
    return best;
  }

  function onPointerDown(event: PointerEvent<HTMLButtonElement>) {
    if (cleared || event.button !== 0) return;
    event.currentTarget.setPointerCapture(event.pointerId);
    dragRef.current = {
      x0: event.clientX,
      y0: event.clientY,
      lastX: event.clientX,
      lastT: event.timeStamp,
      rot: 0,
      moved: false,
    };
  }

  function onPointerMove(event: PointerEvent<HTMLButtonElement>) {
    const drag = dragRef.current;
    if (!drag) return;
    const dx = event.clientX - drag.x0;
    const dy = event.clientY - drag.y0;
    if (!drag.moved) {
      if (Math.hypot(dx, dy) < 4) return;
      drag.moved = true;
      setDragging(true);
      setPicked(false);
    }
    // Lean into the direction of travel, up to 8 degrees.
    const dt = Math.max(event.timeStamp - drag.lastT, 1);
    const lean = reducedMotion()
      ? 0
      : Math.max(-8, Math.min(8, ((event.clientX - drag.lastX) / dt) * 6));
    drag.rot += (lean - drag.rot) * 0.2;
    drag.lastX = event.clientX;
    drag.lastT = event.timeStamp;
    event.currentTarget.style.transform = `translate(${dx}px, ${dy}px) rotate(${drag.rot}deg) scale(1.05)`;
    const target = nearest(event.clientX, event.clientY);
    if (target !== nearRef.current) {
      nearRef.current = target;
      setNear(target);
    }
  }

  function endDrag(event: PointerEvent<HTMLButtonElement>, cancelled: boolean) {
    const drag = dragRef.current;
    if (!drag) return;
    const target = nearRef.current;
    nearRef.current = null;
    setNear(null);
    if (!drag.moved) {
      dragRef.current = null;
      if (!cancelled) setPicked((value) => !value);
      return;
    }
    setDragging(false);
    if (target && !cancelled) drop(target);
    else flyHome(event.currentTarget.style.transform);
    dragRef.current = null;
  }

  // The keyboard path: Enter picks the piece up, focus moves to the first spot, Enter drops it.
  useEffect(() => {
    if (!picked) return;
    const first = slotRefs.current.values().next().value;
    if (first && document.activeElement === pieceRef.current) first.focus();
  }, [picked]);

  function onKeyDown(event: KeyboardEvent) {
    if (event.key === "Escape" && picked) {
      setPicked(false);
      pieceRef.current?.focus();
    }
  }

  return (
    <article
      id={`lv-${n}`}
      className="level panel"
      data-rev
      // Set as it mounts, so the reveal never hides it and the glide back measures it in place.
      data-in={replayed || undefined}
      data-clear={cleared || undefined}
      onKeyDown={onKeyDown}
    >
      <span className="node" aria-hidden="true">
        {cleared ? "✓" : n}
      </span>
      <div className="lv-copy">
        <div className="lv-head">
          <span className="badge">LV {n}</span>
          {cleared && (
            <span className={stars === 0 ? "stamp skip" : "stamp"}>
              {stars === 0 ? "Skipped" : "Clear"}
            </span>
          )}
          {stars !== 0 && <Stars stars={stars} />}
        </div>
        <h3>
          {!cleared && <span className="sr-only">Level {n}, name hidden until cleared.</span>}
          <Scramble text={level.name} shown={cleared} step={26} />
        </h3>
        <p className="prompt">{level.prompt}</p>
        <div
          className="say"
          aria-live="polite"
          data-tone={cleared ? "yes" : said ? "no" : undefined}
        >
          <p key={said ? `${said.id}-${miss?.at ?? 0}` : "hint"}>
            {said ? said.note : "Drag the piece into the tree. Or tap it, then tap a spot."}
          </p>
          {said && (
            <span className="costs">
              {said.cost.map((cost) => (
                <span key={cost}>{cost}</span>
              ))}
            </span>
          )}
        </div>
      </div>

      <div className="board well" data-drag={dragging || picked || undefined}>
        <ul className="tree">
          {level.rows.map((row, index) => {
            if (!("slot" in row)) {
              return (
                <li key={index} className={`row ${row.kind}`} style={{ "--d": row.depth }}>
                  {row.text}
                  {row.calls && <span className="calls">uses it</span>}
                </li>
              );
            }
            const { slot } = row;
            const seated = placed === slot.id;
            return (
              <li key={index} className="row" style={{ "--d": row.depth }}>
                <button
                  type="button"
                  ref={(element) => {
                    if (element) slotRefs.current.set(slot.id, element);
                    else slotRefs.current.delete(slot.id);
                  }}
                  className="slot"
                  disabled={cleared || !(picked || dragging)}
                  data-near={near === slot.id || undefined}
                  data-miss={miss?.id === slot.id || undefined}
                  data-off={(cleared && !seated) || undefined}
                  data-seat={seated || undefined}
                  key={miss?.id === slot.id ? miss.at : "slot"}
                  onClick={() => drop(slot.id)}
                >
                  {seated ? (
                    <span ref={seatRef} className="piece seated">
                      {level.piece}
                      {stars !== 0 && <Burst />}
                    </span>
                  ) : (
                    slot.label
                  )}
                </button>
              </li>
            );
          })}
        </ul>
        <div className="tray">
          {!cleared && (
            <button
              ref={pieceRef}
              type="button"
              className="piece"
              data-picked={picked || undefined}
              data-dragging={dragging || undefined}
              aria-pressed={picked}
              aria-label={`${level.piece}. ${picked ? "Picked up: choose a spot in the tree." : "Pick up"}`}
              onPointerDown={onPointerDown}
              onPointerMove={onPointerMove}
              onPointerUp={(event) => endDrag(event, false)}
              onPointerCancel={(event) => endDrag(event, true)}
              onClick={(event) => {
                // Pointer taps are handled on pointerup; this is Enter and Space.
                if (event.detail === 0) setPicked((value) => !value);
              }}
            >
              {level.piece}
            </button>
          )}
          <span className="tray-note">
            {cleared
              ? "In its place."
              : picked
                ? "Now pick a spot. Esc puts it back."
                : "Your piece"}
          </span>
        </div>
      </div>
    </article>
  );
}

function Burst() {
  return (
    <span className="burst" aria-hidden="true">
      {Array.from({ length: 8 }, (_, index) => (
        <i key={index} style={{ "--a": `${index * 45 + 22}deg` }} />
      ))}
    </span>
  );
}

function CopyMarkdown({ className }: { className: string }) {
  const [copied, setCopied] = useState(false);

  async function handleCopy() {
    try {
      const text = await readMarkdown(MARKDOWN_URL);
      if (text === null) return;
      await navigator.clipboard.writeText(text);
      setCopied(true);
      setTimeout(() => setCopied(false), 1600);
    } catch (e) {
      console.error(e);
    }
  }

  return (
    <button type="button" className={className} onClick={handleCopy}>
      {copied ? "Copied" : "Copy as Markdown"}
    </button>
  );
}

/** Each [data-rev] element rises in once as it enters the screen. Those already on screen when
 * the page mounts are shown at once, so nothing that was painted disappears. */
function useReveal(rootRef: RefObject<HTMLElement | null>) {
  useLayoutEffect(() => {
    const root = rootRef.current;
    if (!root) return;
    const items = [...root.querySelectorAll<HTMLElement>("[data-rev]")];
    for (const item of items) {
      if (item.getBoundingClientRect().top < innerHeight) item.dataset.in = "";
    }
    root.dataset.revOn = "";
    const observer = new IntersectionObserver(
      (entries) => {
        for (const entry of entries) {
          if (!entry.isIntersecting || !(entry.target instanceof HTMLElement)) continue;
          entry.target.dataset.in = "";
          observer.unobserve(entry.target);
        }
      },
      { rootMargin: "0px 0px -10% 0px" },
    );
    for (const item of items) if (!("in" in item.dataset)) observer.observe(item);
    return () => observer.disconnect();
  }, [rootRef]);
}
